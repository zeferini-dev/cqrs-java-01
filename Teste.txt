package com.example.repository;

import java.util.List;

public interface CustomRepository {
    <T> List<T> executeListQuery(String sql, Class<T> type);

    boolean executeUpdateQuery(String sql);

    Integer executeCountQuery(String sql);
}

// Implementação do Repositório Customizado
// Arquivo: src/main/java/com/example/repository/CustomRepositoryImpl.java
package com.example.repository;

import jakarta.persistence.EntityManager;
import jakarta.persistence.Query;
import org.springframework.stereotype.Repository;
import java.util.List;

@Repository
public class CustomRepositoryImpl implements CustomRepository {

    private final EntityManager entityManager;

    public CustomRepositoryImpl(EntityManager entityManager) {
        this.entityManager = entityManager;
    }

    @Override
    public <T> List<T> executeListQuery(String sql, Class<T> type) {
        try {
            Query query = entityManager.createNativeQuery(sql, type);
            return query.getResultList();
        } catch (Exception e) {
            throw new RuntimeException("Error executing list query: " + e.getMessage(), e);
        }
    }

    @Override
    public boolean executeUpdateQuery(String sql) {
        try {
            Query query = entityManager.createNativeQuery(sql);
            query.executeUpdate();
            return true;
        } catch (Exception e) {
            return false;
        }
    }

    @Override
    public Integer executeCountQuery(String sql) {
        try {
            Query query = entityManager.createNativeQuery(sql);
            Object result = query.getSingleResult();
            return result != null ? Integer.parseInt(result.toString()) : 0;
        } catch (Exception e) {
            throw new RuntimeException("Error executing count query: " + e.getMessage(), e);
        }
    }
}

// Controlador REST
// Arquivo: src/main/java/com/example/controller/MyEntityController.java
package com.example.controller;

import com.example.dto.CustomJoinResult;
import com.example.repository.MyEntityRepository;
import org.springframework.web.bind.annotation.*;

import java.util.List;

@RestController
@RequestMapping("/api/entities")
public class MyEntityController {

    private final MyEntityRepository repository;

    public MyEntityController(MyEntityRepository repository) {
        this.repository = repository;
    }

    @GetMapping("/joins")
    public List<CustomJoinResult> getJoinResults() {
        String sql = """
                SELECT e.id AS id, e.name AS name, b.data AS relatedData
                FROM entity_a e
                JOIN entity_b b ON e.related_id = b.id
                """;
        return repository.executeListQuery(sql, CustomJoinResult.class);
    }

    @PostMapping("/update")
    public boolean updateEntity(@RequestParam String sql) {
        return repository.executeUpdateQuery(sql);
    }

    @GetMapping("/count")
    public Integer getCount(@RequestParam String sql) {
        return repository.executeCountQuery(sql);
    }
}
